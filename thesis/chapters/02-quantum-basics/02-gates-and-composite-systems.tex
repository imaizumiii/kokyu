\section{量子ゲートと複数量子ビット}
\label{sec:gates-composite}

\subsection{ユニタリ変換と1量子ビットゲート}

量子状態に線形写像を作用させることにより，量子状態を変化させることができる．ただし，変化後の量子状態も長さが$1$に規格化されていなければならない．したがって，量子状態の変化はユニタリ写像で表される．

\begin{prop}
作用前の量子状態の長さが$1$であるとき，ユニタリ写像$U$を作用させた後の量子状態の長さも$1$である．
\end{prop}
\begin{proof}
長さが$1$に規格化された任意の量子状態$\ket{\psi}$について，任意のユニタリ写像$U$を作用させると，作用後の量子状態は$U\ket{\psi}$である．そのノルムの二乗は
\[
  \lVert U\ket{\psi}\rVert^2
  =\bra{\psi}U^\dagger U\ket{\psi}
  =\braket{\psi|\psi} = 1
\]
である．
また，$U^\dagger U=UU^\dagger=I$であるので，$U^\dagger$は$U$の逆写像である．
\end{proof}

基本的な1量子ビットに対するユニタリ写像を
\begin{align*}
  X&\coloneq\begin{bmatrix}0&1\\1&0\end{bmatrix},&
  Y&\coloneq\begin{bmatrix}0&-\mathrm{i}\\\mathrm{i}&0\end{bmatrix},&
  Z&\coloneq\begin{bmatrix}1&0\\0&-1\end{bmatrix},\\
  H&\coloneq\frac1{\sqrt2}\begin{bmatrix}1&1\\1&-1\end{bmatrix}
\end{align*}
で定める．特に，アダマールゲートと呼ばれる$H$を$\ket{0}, \ket{1}$に作用させたものを$\ket{+}, \ket{-}$と書き，
\begin{equation}
  \ket{+}\coloneq H\ket{0}=\frac{\ket{0}+\ket{1}}{\sqrt2},
  \qquad
  \ket{-} \coloneq H\ket{1}=\frac{\ket{0}-\ket{1}}{\sqrt2}
  \label{eq:hadamard-action}
\end{equation}
と定める．これらは互いに直交し，$\{\ket{+},\ket{-}\}$は正規直交基底をなす．$\{\ket{+},\ket{-}\}$を\textbf{アダマール基底}という．

後で用いる$y$軸回りの回転ゲートを
\[
  R_y(\theta)\coloneq
  \begin{bmatrix}
    \cos(\theta/2)&-\sin(\theta/2)\\
    \sin(\theta/2)& \cos(\theta/2)
  \end{bmatrix}
\]
と定める．このとき，
\begin{equation}
  R_y(\theta)\ket{0}
  =\cos(\theta/2)\ket{0}+\sin(\theta/2)\ket{1}
  \label{eq:ry-on-zero}
\end{equation}
が成り立つ．

\subsection{テンソル積と複合系}

二つの量子ビットを同時に扱うには，二つのベクトル空間を組み合わせる必要がある．
この組合せに使う演算が\textbf{テンソル積}である．

\begin{defi}[行列のテンソル積]
$A=(a_{ij})\in\mathbb{C}^{m\times n}$，$B\in\mathbb{C}^{r\times s}$に対し
\begin{equation}
 A\otimes B
 :=
 \begin{bmatrix}
  a_{11}B & a_{12}B & \cdots & a_{1n}B\\
  a_{21}B & a_{22}B & \cdots & a_{2n}B\\
  \vdots  & \vdots  & \ddots & \vdots\\
  a_{m1}B & a_{m2}B & \cdots & a_{mn}B
 \end{bmatrix}
 \label{eq:matrix-tensor-product-definition}
\end{equation}
を$A$と$B$のテンソル積という．これは$mr\times ns$行列である．
\end{defi}

\begin{theo}[テンソル積の性質]
$m,n,r,s,p,t,u,v$を正の整数とし，
\[
\begin{gathered}
  A,A'\in\mathbb{C}^{m\times n},\qquad
  B,B'\in\mathbb{C}^{r\times s}, \qquad
  C\in\mathbb{C}^{n\times p},\qquad
  D\in\mathbb{C}^{s\times t},\qquad
  E\in\mathbb{C}^{u\times v}
\end{gathered}
\]
とする．また，
$\ket{x},\ket{x'}\in\mathbb{C}^{n}$，
$\ket{y},\ket{y'}\in\mathbb{C}^{s}$とする．
このとき，次が成り立つ．

\begin{equation}
  \begin{aligned}
    (A+A')\otimes B&=A\otimes B+A'\otimes B,\\
    A\otimes(B+B')&=A\otimes B+A\otimes B'.
  \end{aligned}
\end{equation}
\begin{equation}
  (A\otimes B)\otimes E=A\otimes(B\otimes E).
\end{equation}
\begin{equation}
  (A\otimes B)(C\otimes D)=(AC)\otimes(BD).
\end{equation}
\begin{equation}
  (A\otimes B)^\dagger=A^\dagger\otimes B^\dagger.
\end{equation}
\end{theo}
\begin{proof}
$B=(b_{k\ell})$，$B'=(b'_{k\ell})$とおく．
行と列をそれぞれ組$(i,k)$，$(j,\ell)$で表すと，
テンソル積の定義から
\[
  (A\otimes B)_{(i,k),(j,\ell)}=a_{ij}b_{k\ell}
\]
である．各成分は複素数なので，分配律が成り立ち次のように計算できる．
\begin{align*}
  \bigl((A+A')\otimes B\bigr)_{(i,k),(j,\ell)}
    &=(a_{ij}+a'_{ij})b_{k\ell}\\
    &=a_{ij}b_{k\ell}+a'_{ij}b_{k\ell}\qquad (\because \text{分配律})\\
    &=\bigl(A\otimes B+A'\otimes B\bigr)_{(i,k),(j,\ell)},\\
  \bigl(A\otimes(B+B')\bigr)_{(i,k),(j,\ell)}
    &=a_{ij}(b_{k\ell}+b'_{k\ell})\\
    &=a_{ij}b_{k\ell}+a_{ij}b'_{k\ell}\qquad (\because \text{分配律})\\
    &=\bigl(A\otimes B+A\otimes B'\bigr)_{(i,k),(j,\ell)}
\end{align*}
となる．

$E=(e_{ab})$とおく．両辺の$(i,k,a)$行$(j,\ell,b)$列の成分を比べると，
\begin{align*}
  \bigl((A\otimes B)\otimes E\bigr)_{(i,k,a),(j,\ell,b)}
    &=(a_{ij}b_{k\ell})e_{ab}\\
    &=a_{ij}(b_{k\ell}e_{ab})\qquad (\because \text{結合律})\\
    &=\bigl(A\otimes(B\otimes E)\bigr)_{(i,k,a),(j,\ell,b)}
\end{align*}
である．

$C=(c_{jh})$，$D=(d_{\ell q})$とおくと，
行列積の定義より
\begin{align*}
  \bigl((A\otimes B)(C\otimes D)\bigr)_{(i,k),(h,q)}
    &=\sum_{j=1}^{n}\sum_{\ell=1}^{s}
      a_{ij}b_{k\ell}c_{jh}d_{\ell q}\\
    &=\left(\sum_{j=1}^{n}a_{ij}c_{jh}\right)
      \left(\sum_{\ell=1}^{s}b_{k\ell}d_{\ell q}\right)\\
    &=(AC)_{ih}(BD)_{kq}\\
    &=\bigl((AC)\otimes(BD)\bigr)_{(i,k),(h,q)}
\end{align*}
となる．最後に，共役転置の定義から
\begin{align*}
  \bigl((A\otimes B)^\dagger\bigr)_{(j,\ell),(i,k)}
    &=\overline{a_{ij}b_{k\ell}}\\
    &=\overline{a_{ij}}\,\overline{b_{k\ell}}\\
    &=(A^\dagger)_{ji}(B^\dagger)_{\ell k}\\
    &=\bigl(A^\dagger\otimes B^\dagger\bigr)_{(j,\ell),(i,k)}
\end{align*}
である．それぞれ対応する成分が一致するので，定理の各等式が従う．
\end{proof}
複数の量子系を合わせたものを複合量子系という．
複合量子系の状態空間は，各量子系の状態空間のテンソル積で表す．
$\ket{i}\otimes\ket{j}$のことを省略して$\ket{ij}$と書く．
たとえば2量子ビットの計算基底は
\[
\ket{00},\ket{01},\ket{10},\ket{11}
\]
である．本論文では左側の因子を上位の量子ビットとし，
$\ket{x_{n-1}\cdots x_0}$を整数$\sum_{j=0}^{n-1}2^jx_j$に対応させる．

\subsection{複数量子ビットに作用するゲート}

\begin{defi}[制御ユニタリ]
一方を制御量子ビット，もう一方を標的とする．標的の状態を表すベクトル空間を$\mathcal K$とする．このとき，
\begin{equation}
 C(U)=\ket{0}\bra{0}\otimes I_{\mathcal K}
      +\ket{1}\bra{1}\otimes U
 \label{eq:controlled-unitary}
\end{equation}
を制御$U$演算という．
\end{defi}

\begin{prop}
写像$U$がユニタリなら$C(U)$もユニタリである．
\end{prop}
\begin{proof}
射影を$P_0=\ket{0}\bra{0}$，$P_1=\ket{1}\bra{1}$とおくと，
$P_iP_j=\delta_{ij}P_i$，$P_0+P_1=I$である．よって
\begin{align*}
 C(U)^\dagger C(U)
 &=(P_0\otimes I+P_1\otimes U^\dagger)
   (P_0\otimes I+P_1\otimes U)\\
 &=P_0\otimes I+P_1\otimes U^\dagger U
  =(P_0+P_1)\otimes I=I
\end{align*}
である．同様に$C(U)C(U)^\dagger=I$も得られる．
\end{proof}

標的に作用する演算を$U=X$としたものがCNOTゲートである．
また，制御量子ビットが$\alpha\ket{0}+\beta\ket{1}$であり，
標的の状態が$\ket{\psi}$であるとする．線形性より
\begin{equation}
 C(U)(\alpha\ket{0}+\beta\ket{1})\ket{\psi}
 =\alpha\ket{0}\ket{\psi}+\beta\ket{1}U\ket{\psi}
 \label{eq:controlled-unitary-superposition}
\end{equation}
となる．この式は位相推定と制御回転で繰り返し用いる．
